{{- /* Markdown 图片渲染钩子。三件事： 1. 让照片可以点击放大（灯箱与相册页共用同一套样式） 2. 按文件名自动查出拍摄参数，显示在图注下面 3. 支持排版尺寸指令，让图文错落有致 用法还是最普通的 Markdown： ![替代文字](图片地址) ![替代文字](图片地址 "图注文字") ![替代文字](图片地址 "[wide] 图注文字") 可选的大小指令（写在图注最前面，方括号里）： （不写） 与正文同宽 —— 最稳妥的默认 [wide] 出血，比正文宽一些 —— 适合风景 [full] 占满整个屏幕宽度 —— 强调用，一张文章别超过两张 [small] 缩到 62% 居中 —— 适合特写、竖图 [half] 缩到 46% 居中 —— 更小的细节图 注意：这里只输出行内元素（a / img / span），不包 figure。 因为 Markdown 里独占一段的图片会被包进

，里面放块级元素是非法 HTML。 */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- /* 解析开头的尺寸指令 */ -}} {{- $size := "" -}} {{- $caption := $title -}} {{- range $s := slice "wide" "full" "small" "half" -}} {{- if hasPrefix $title (printf "[%s]" $s) -}} {{- $size = $s -}} {{- $caption = replaceRE `^\[[a-z]+\]\s*` "" $title -}} {{- end -}} {{- end -}} {{- /* 解析图片地址（中英文共用同一个图片文件夹，这里会自动找到正确的路径）*/ -}} {{- $src := partial "image-url.html" (dict "Page" .Page "Path" $dest) -}} {{- /* 灯箱指向「完整分辨率」的那一份：foo.webp → foo-full.webp。 上传脚本会给每张照片同时生成两份，万一某张没有，JS 会自动退回正文图。 */ -}} {{- $full := $src -}} {{- if findRE `\.(webp|jpe?g)$` $src -}} {{- $full = replaceRE `\.(webp|jpe?g)$` "-full.$1" $src -}} {{- end -}} {{- /* 按文件名查拍摄参数。表由 scripts/exif.py 生成，上传照片时自动更新。 */ -}} {{- $exif := index (index site.Data "photo-exif" | default dict) (path.Base $src) -}} {{- $exifLine := "" -}} {{- with $exif -}} {{- $bits := slice -}} {{- with .aperture }}{{ $bits = $bits | append . }}{{ end -}} {{- with .shutter }}{{ $bits = $bits | append . }}{{ end -}} {{- with .iso }}{{ $bits = $bits | append (printf "ISO %s" .) }}{{ end -}} {{- with .focal }}{{ $bits = $bits | append . }}{{ end -}} {{- $exifLine = delimit $bits " · " -}} {{- end -}} {{- /* 一页只引入一次灯箱的样式和脚本 */ -}} {{- if not (.Page.Store.Get "postLightboxLoaded") -}} {{- .Page.Store.Set "postLightboxLoaded" true -}} {{- end -}} {{ with $caption }}{{ . }}{{ end }}{{ with $exifLine }}{{ . }}{{ end }}
